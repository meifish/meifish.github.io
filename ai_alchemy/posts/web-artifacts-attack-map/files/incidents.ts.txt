// 2026 AI attack incidents — hand-curated from public reporting (as of 2026-10-31).
// Rule of the map: an *attribution* pin is where an investigator says the actor
// (or its sponsor/employer) is. A *traffic* pin is where the packets were seen
// coming from. They are different things and are drawn differently.
// Coordinates are city/capital/region centroids, never a hacker's actual desk.

export type Confidence = "high" | "medium" | "low" | "none";
export type Category = "espionage" | "crime" | "ai-target" | "agent-mishap";

export interface Place {
  name: string;
  lat: number;
  lon: number;
  approx?: string; // why this point is only approximate
}

export interface Source {
  name: string;
  url: string;
  primary?: boolean; // the investigator's own write-up, read in full
}

export interface Incident {
  id: string;
  title: string;
  when: string; // human readable
  month: number; // 1-12 of 2026, month the activity was disclosed/observed (for the timeline)
  category: Category;
  impact: 1 | 2 | 3 | 4 | 5;
  impactWhy: string;
  aiRole: string;
  summary: string;
  attribution: {
    claim: string; // exact-ish wording
    by: string; // who made the call
    confidence: Confidence;
    place: Place | null; // null = no location claimed
  };
  traffic: {
    note: string; // what is known about where traffic came from
    places: Place[];
  };
  targets: Place[]; // empty = global / not disclosed
  targetNote: string;
  sources: Source[];
}
